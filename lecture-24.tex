\subsection{Stokes' Theorem and the Divergence Theorem}

\subsubsection{Stokes' Theorem}

\content{
\begin{theorem} Let $S$ be an oriented piecewise-smooth surface that is bounded
by a simple closed piecewise-smooth boundary curve $C$ with positive
orientation. Let $\myvec{F}$ be a vector field whose components have continuous
partial derivatives on an open region in $\R^3$ that contains $S$. Then 
$$ \int_C \myvec{F}\cdot d\myvec{r} = \iint_S \text{curl}\, \myvec{F}\cdot d\myvec{S}.
$$
\end{theorem}

-y^2\myvec{i} +x\myvec{j} +z^2\myvec{k}$ and $C$ is the curve of intersection of the
plane $y+z=2$ and the cylinder $x^2+y^2=1$.
\end{example}

\subsubsection{The Divergence Theorem}


\content{
\begin{theorem} Let $E$ be a simple solid region and let $S$ be the boundary
surface of $E$, given with positive orientation. Let $\myvec{F}$ be a vector field
whose component functions have continuous partial derivatives on an open region
containing $E$, then 
$$ \iint_S \myvec{F}\cdot d\myvec{S} = \iiint_E \text{div}\,\myvec{F}\, dV. $$
\end{theorem}
}{% presentation
}{% nopresentation
}{% comments

\end{example}
