\subsection{Curl and Divergence}


\content{
\begin{definition} Let $\myvec{F} = P\myvec{i} + Q\myvec{j} +
R\myvec{k}$ be a vector field on $\R^3$ and suppose that the partial
derivatives of $P$, $Q$, and $R$ all exist, then the curl of $\myvec{F}$ is
another vector field on $\R^3$ defined by 
$$ \text{curl}\,\myvec{F} = \left(\pd{P}{y}-\pd{Q}{z}\right)\myvec{i} + 
\left(\pd{P}{z}-\pd{R}{x}\right)\myvec{j} + 
\left(\pd{Q}{x}-\pd{P}{y}\right)\myvec{k}. $$
\end{definition}
}{% presentation
\vspace{3in}
}{% nopresentation
}{% comments
To aid our memory, introduce the vector differential operator ``del'' 
$$ \nabla = \pd{}{x}\myvec{i} + \pd{}{y}\myvec{j} + \pd{}{z}\myvec{k}. $$
This acts on a scalar function to produce the gradient of $f$. Then we can think
of 
$$ \text{curl}\,\myvec{F} = \nabla\times \myvec{F}. $$
}


\content{
\begin{example} If $\myvec{F}(x,y,z) = xy\myvec{i} + xyz\myvec{j} -y^2\myvec{k}$, find 
$\text{curl}\,\myvec{F}$. 
\end{example}
}{% presentation
\vspace{3in}
}{% nopresentation
}{% comments
}

\content{
\begin{theorem} If $f$ is a function of three variables that has continuous
second-order derivatives, then $\text{curl}(\nabla f)=\myvec{0}$.
\end{theorem}
}{% presentation
\vspace{3in}
}{% nopresentation
}{% comments
prove this quickly. Use this to show that if a vector field is conservative,
then it's curl is zero. 
}

\content{
\begin{example} Show that the vector field $\myvec{F}(x,y,z) = xz\myvec{i} +
xyz\myvec{j} -y^2\myvec{k}$ is not conservative. 
\end{example}
}{% presentation
}{% nopresentation
}{% comments
By previous comments, this is trivial, since the curl of this vector field was
computed above and is not zero. 
}

\content{
\begin{theorem} If $\myvec{F}$ is a vector field defined on all of $\R^3$ whose
component functions have continous partial derivatives and
$\text{curl}\,\myvec{F}=\myvec{0}$, then $\myvec{F}$ is a conservative vector field. 
\end{theorem}

\content{
\begin{definition} If $\myvec{F}=P\myvec{i} + Q\myvec{j} + R\myvec{k}$ is a vector field
on $\R^3$, we define the divergence of $\myvec{F}$ to be 
$$ \text{div}\,\myvec{F} = \pd{P}{x} + \pd{Q}{y} + \pd{R}{z}, $$
provided the above partial derivatives exist.
\end{definition}
}{% presentation
\vspace{0.5in}
}{% nopresentation
}{% comments
Mention that this can be remembered by writing
$$ \nabla \cdot \myvec{F}$$. 
}

\end{example}

\content{
\begin{theorem} If $\myvec{F} = P\myvec{i} + Q\myvec{j} + R\myvec{k}$ is a vector field
on $\R^3$ and $P$, $Q$, and $R$ have continuous second order partial
derivatives, then 
$$ \normalfont\text{div}\,\text{curl}\,\myvec{F} = 0. $$
\end{theorem}
}{% presentation
}{% nopresentation
}{% comments
}

\content{

-y^2\myvec{k}$ can't be written as the curl of another vector field, that is,
$\myvec{F}\neq \text{curl}\,\myvec{G}$ for any vector field $\myvec{G}$.  
\end{example}
}{% presentation
\vspace{2in}
}{% nopresentation
}{% comments
