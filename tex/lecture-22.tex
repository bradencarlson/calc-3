\subsection{Applications of Double Integrals}

\content{
        \begin{definition} If $D$ is a (completely flat) plate, and $f(x,y)$ denotes the mass 
        $D$ at the point $(x,y)$, the total mass of $D$ is simply 
        $$ \iint_D f(x,y) dA. $$
        \end{definition}
}{% presentation
}{% nopresentation
}{% comments
}


\content{
        \begin{definition} Suppose that we have a (completely flat) lamina which
        occupies the region $D$ and whose mass at point $(x,y)$ is
        $\rho(x,y)$. Its center of mass is given by the
        point $(\overline{x}, \overline{y})$ where 
        $$ m\overline{x} = \iint_D x \rho(x,y) dA,\quad \text{and} \quad
        m\overline{y} = \iint_D y\rho(x,y) dA, $$
        where $m$ is the total mass of the lamina.
        \end{definition}
}{% presentation
}{% nopresentation
}{% comments
}


\content{
        \begin{example}
                Find the total mass and center of mass of the lamina which
                occupies the triangular region of space with vertices $(0,0)$,
                $(1,0)$, and $(0,2)$ and whose mass at point $(x,y)$ is given by 
                $ \rho(x,y) = 1+3x+y$.
        \end{example}
}{% presentation
        \vspace{4in}
}{% nopresentation
        \vspace{3in}
}{% comments
        mass is $8/3$. center of mass is $(3/8, 11/16)$.

        Mention that there are a few other applications in the book if you want
        to look them over. 
}

\subsection{Surface Area}


\content{
}{% presentation
        \vspace{4in}
}{% nopresentation
        \vspace{3in}
}{% comments
        Derive the formula for surface area:
        $$ \iint_D \sqrt{1 + f_x^2 + f_y^2} dA $$
}


\content{
        \begin{example} Find the surface area of the surface 
                $$ z = x^2 +2y+2 $$
        over the triangular region with vertices $(0,0)$ $(1,0)$ and $(0,1)$.
        \end{example}
}{% presentation
        \vspace{3in}
}{% nopresentation
        \vspace{2in}
}{% comments
        $$ \frac{1}{12} ( 27 - 5\sqrt{5}) $$
}

\content{
        \begin{example} Find the surface area of the surface $z=x^2+y^2$ which
                lies beneath the plane $z=9$.
        \end{example}
}{% presentation
        \vspace{3in}
}{% nopresentation
        \vspace{2in}
}{% comments
        $$ \frac{\pi}{6}(37\sqrt{37} - 1) $$
}
