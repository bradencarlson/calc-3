\subsection{Change of Variables in Multiple Integrals}


\content{

}{% presentation
\vspace{3in}
}{% nopresentation
}{% comments
Recall that for single integrals, we have the change of variable formula
$$ \int f(x)\,dx = \int f(g(u))g'(u)\,du. $$
We wish to have a similar formula in multiple integrals. First discuss 2
variable case, where we deal with transformations of the form $T(u,v) = (x,y)$. 
We want a formula like the following
$$ \iint f(x,y)\, dA_{xy} = \iint f(T(u,v))h(u,v)\, dA_{uv}, $$
where we currently do not know what $h(u,v)$ is. 
}

\content{
\begin{definition} The Jacobian of the transformation $T$, given by $x = g(u,v)$
and $y=h(u,v)$ is 
$$ \pd{(x,y)}{(u,v)} = \left|\begin{matrix}
\pd{x}{u} & \pd{x}{v} \\
\pd{y}{u} & \pd{y}{v} \end{matrix}\right|= 
\pd{x}{u}\pd{y}{v} - \pd{x}{v}\pd{y}{u}. $$
\end{definition}
}{% presentation
}{% nopresentation
}{% comments
}

\content{
\begin{theorem} (Change of Variables in Double Integrals) Suppose that $T$ is a
transformation given by $x=g(u,v)$ and $y=h(u,v)$, where both $g$ and $h$ have
continuous first partial derivatives, and that $T$ maps the region $S$ in the
$uv$-plane to the region $R$ in the $xy$-plane.  Suppose that $f$ is continuous
on $R$ and that $S$ and $R$ are both of type I or type II. Suppose further that
$T$ is one-to-one, except possibly on the boundary of $S$. Then 
$$ \\int_R f(x,y)\, dA_{xy} = \iint_S
f(g(u,v),h(u,v))\left|\pd{(x,y)}{(u,v)}\right|\,dA_{uv}. $$
\end{theorem}
}{% presentation
}{% nopresentation
}{% comments
}


\content{
\begin{example} (Polar Coordinates) Define $T$ from the $r\theta$-plane to
the $xy$-plane by $x=r\cos\theta$ and $y=r\sin\theta$. Find the change of
variables formula for this transformation. 
\end{example}
}{% presentation
\vspace{3in}
}{% nopresentation
}{% comments
}

\subsubsection{Change of Variables in Triple Integrals}

\content{
Let $T$ be a one-to-one transformation from $uvw$-space to $xyz$-space defined
by the equations 
$$ x = g(u,v,w)\quad y=h(u,v,w)\quad z=k(u,v,w). $$
\begin{definition} The Jacobian of $T$ is 
$$ \pd{(x,y,z)}{(u,v,w)}= \left|\begin{matrix} 
\pd{x}{u} & \pd{x}{v} & \pd{x}{w} \\
\pd{y}{u} & \pd{y}{v} & \pd{y}{w} \\
\pd{z}{u} & \pd{z}{v} & \pd{z}{w} \end{matrix}\right|. $$
\end{definition}
This allows us to obtain the following change of variables formula for triple
integrals
$$ \iiint_R f(x,y,z)\, dA_{xyz} = \iiint_S
f(T(u,v,w))\left|\pd{(x,y,z)}{(u,v,w)}\right|\, dA_{uvw}. $$
