\subsection{Maximum and Minimum Values}

\content{
\begin{definition} If $f$ is a function of two variables, $x$ and $y$, we say
that $f$ has a (local) minimum at $(a,b)$ if $(a,b)$ lies in the domain of $f$,
and for points $(x,y)$ in the domain of $f$ 'near` $(a,b)$ we have that 
$$ f(a,b) \leq f(x,y). $$ 
Similarly, we say that $(a,b)$ is a (local) maximum if 
$$ f(a,b)\geq f(x,y) $$
for all $(x,y)$ 'near` $(a,b)$ and in the domain of $f$. 
\end{definition}

}{% comments

\content{
\begin{theorem} If $f$ is a function of two variables, and has local minimum or
maximum at $(a,b)$, and the partial derivatives of $f$ exist at $(a,b)$, we have
that 
$$ \pd{f}{x}(a,b) =0 \text{ and } \pd{f}{y}(a,b) = 0. $$
\end{theorem}

\content{
\begin{theorem} (Second derivative test) Suppose that $f$ is a function of two
variables, whose second partial derivatives are continuous around the critical
point $(a,b)$. Then we define 
$$ D(a,b) = \left|\begin{matrix} f_{xx}(a,b) & f_{xy}(a,b) \\
f_{xy}(a,b) & f_{yy}(a,b) \end{matrix}\right|, $$
and we have the following 
\begin{enumerate}
\item if $D(a,b)> 0$ and $f_{xx}(a,b)>0$, then $f(a,b)$ is a local minimum. 
\item If $D(a,b)>0$ and $f_{xx}(a,b)<0$, then $f(a,b)$ is a local maximum. 
\item If $D(a,b)< 0$, then $f(a,b)$ is a saddle point. 
\end{enumerate}
\end{theorem}

}{% nopresentation
}{% comments

\content{
\begin{definition} If $f$ is a function of two variables, we say that $f$ has a
global (or absolute) minimum at $(a,b)$ if $f(x,y) \geq f(a,b)$ for all $(x,y)$
in the domain of $f$. Similarly, $f$ has a global maximum at $(a,b)$ if 
$f(x,y) \leq f(a,b) $ for all $(x,y)$ in the domain of $f$. 
\end{definition}

\content{
\begin{theorem} (Extreme Value Theorem) If $f$ is a continuous function of two
variables, defined on a closed and bounded set $D$, then $f$ attains an absolute
maximum and minimum value on $D$.
\end{theorem}
}{% presentation
}{% nopresentation
}{% comments
This lets us define a method similar to the 'closed interval method' from calc I
}

\content{
To find the maximum and minimum values of a continuous function $f$ on a closed
and bounded set $D$, we do the following
\begin{enumerate}
\item Find the values of $f$ at the critical points of $f$ in $D$. 
\item Find the extreme values of $f$ on the boundary of $D$.
\item Identify the largest and smallest of these values. 
\end{enumerate}
