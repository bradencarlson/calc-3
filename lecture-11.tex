\subsection{Functions of Several Variables}

\content{
\begin{definition} A function of two variables is a rule which assigns to
each ordered pair of real numbers $(x,y)$ from a given domain $D$, which is
denoted $f(x,y)$. The set $D$ is known as the domain, and it's range is the set
of real values obtained as output of $f$.
\end{definition}
}{% presentation
}{% nopresentation
}{% comments
$x$ and $y$ are known as the independent variables, and $z=f(x,y)$ is the
dependent variable. 
}

\content{
\begin{example}
For the following function, evaluate $f(3,2)$ and find and sketch the domain.
$$ f(x,y) = \frac{\sqrt{x+y+1}}{x-1}. $$
\end{example}
}{% presentation
\vspace{4in}
}{% nopresentation
}{% comments
}

\content{
\begin{example} Find the domain of the function 
$$ f(x,y) = \sqrt{9-x^2-y^2}. $$
\end{example}

\subsubsection{Graphs}

\content{
\begin{definition} If $f$ is a function of two variables with domain $D$, then
the graph of $f$ is the set of all points $(x,y,z)$ in $\R^3$ such that
$z=f(x,y)$ and $(x,y)$ is in $D$.
\end{definition}
}{% presentation
}{% nopresentation
}{% comments
}

\content{
\begin{example}
Sketch the graph of the function $f(x,y)=6-3x-2y$.
\end{example}
}{% presentation
\vspace{3in}
}{% nopresentation

plane, simply draw the triangle between these three points.
